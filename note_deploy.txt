Phần 2: build-docker có: nginx và file default.conf 

# Redirect HTTP → HTTPS
server {
    listen 80;
    server_name vnptbacninh.com www.vnptbacninh.com;
    return 301 https://$host$request_uri;
}

# HTTPS server
server {
    listen 443 ssl;
    server_name vnptbacninh.com www.vnptbacninh.com;

    ssl_certificate /etc/letsencrypt/live/vnptbacninh.com/fullchain.pem;
    ssl_certificate_key /etc/letsencrypt/live/vnptbacninh.com/privkey.pem;

    root /usr/share/nginx/html;
    index index.html;

    location / {
        try_files $uri $uri/ /index.html;
    }

	location /api/ {
		proxy_pass http://backend-spring:8080/api/;
		proxy_set_header Host $host;
		proxy_set_header X-Real-IP $remote_addr;
		proxy_set_header X-Forwarded-For $proxy_add_x_forwarded_for;
		proxy_set_header X-Forwarded-Proto $scheme;
	}
	
	location /storage/ {
        alias /uploads/;                    # 👈 Trỏ tới thư mục mount trong container nginx
        autoindex on;                       # (tùy chọn, cho phép duyệt file)
        access_log off;
    }

    error_page 500 502 503 504 /50x.html;
    location = /50x.html {
        root /usr/share/nginx/html;
    }
}


Phần 3: Dockerfile trong smart-java-spring-final:
# Stage 1: Build the application
FROM gradle:8.7-jdk17 AS build
COPY --chown=gradle:gradle . /project/smart
WORKDIR /project/smart

#skip task: test
RUN gradle clean build -x test --no-daemon

# Stage 2: Run the application
FROM openjdk:17-slim
EXPOSE 8080
COPY --from=build /project/smart/build/libs/*.jar /project/spring-boot-project.jar
ENTRYPOINT ["java", "-jar", "/project/spring-boot-project.jar"]

version: "3.9"

services:
  nginx:
    image: nginx:1.23
    container_name: smart-nginx
    volumes:
      - ../smart-react-vite-final/dist:/usr/share/nginx/html
      - ./nginx/default.conf:/etc/nginx/conf.d/default.conf
      - ../upload-docker/:/uploads/    
      - /etc/letsencrypt/live/vnptbacninh.com:/etc/letsencrypt/live/vnptbacninh.com:ro
      - /etc/letsencrypt/archive/vnptbacninh.com:/etc/letsencrypt/archive/vnptbacninh.com:ro
    ports:
      - "80:80"
      - "443:443"
    depends_on:
      - backend-spring
    restart: always
    networks:
      - smart-network

  db-mysql:
    image: mysql:5.7.40
    container_name: smart-db
    restart: always
    environment:
      - MYSQL_DATABASE=smart
      - MYSQL_ROOT_PASSWORD=123456
    ports:
      - "3307:3306"
    expose:
      - "3306"
    volumes:
      - mysql-data:/var/lib/mysql
    healthcheck:
      test: ["CMD", "mysqladmin", "ping", "-h", "localhost"]
      interval: 10s
      timeout: 5s
      retries: 5
    networks:
      - smart-network

  backend-spring:
    build:
      context: ../smart-java-spring-final
      dockerfile: Dockerfile
    container_name: smart-backend
    ports:
      - "8088:8080"
    environment:
      - SPRING_DATASOURCE_URL=jdbc:mysql://db-mysql:3306/smart
      - SPRING_DATASOURCE_USERNAME=root
      - SPRING_DATASOURCE_PASSWORD=123456
      - SPRING_JPA_HIBERNATE_DDL_AUTO=update
      - PROJECT_UPLOAD_FILE_BASE_URI=file:///uploads/
    volumes:
      - ../upload-docker/:/uploads/
    depends_on:
      - db-mysql
    restart: always
    networks:
      - smart-network

volumes:
  mysql-data:

networks:
  smart-network:
    driver: bridge
