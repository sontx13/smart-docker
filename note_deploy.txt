Phần 2: build-docker có: nginx và file default.conf 

upstream backend-spring-rest {
        server backend-spring:8080;
}

server {
    listen 80;
    server_name vnptbacninh.com;

    location / {
        root /usr/share/nginx/html;
        try_files $uri $uri/ /index.html;
    }

    location /api {
        proxy_pass http://backend-spring-rest;
        proxy_set_header X-Real-IP $remote_addr;
        proxy_set_header X-Forwarded-For $proxy_add_x_forwarded_for;
        proxy_set_header Host $host;
        proxy_set_header X-NginX-Proxy true;
    }

    location /storage {
        try_files $uri $uri/ @proxy;
    }
    location @proxy {
        proxy_pass http://backend-spring-rest;
    }

}

Phần 2: Dockerfile trong smart-java-spring-final:
# Stage 1: Build the application
FROM gradle:8.7-jdk17 AS build
COPY --chown=gradle:gradle . /project/smart
WORKDIR /project/smart

#skip task: test
RUN gradle clean build -x test --no-daemon

# Stage 2: Run the application
FROM openjdk:17-slim
EXPOSE 8080
COPY --from=build /project/smart/build/libs/*.jar /project/spring-boot-project.jar
ENTRYPOINT ["java", "-jar", "/project/spring-boot-project.jar"]

Phần 4: docker-compose.yml trong 

services:
  nginx:
    image: nginx:1.23
    volumes:
      - ../smart-react-vite-final/dist:/usr/share/nginx/html
      - ./nginx/default.conf:/etc/nginx/conf.d/default.conf

    ports:
      - "80:80"
      - "443:443"
  db-mysql:
    image: mysql:5.7.40
    restart: always
    environment:
      - MYSQL_DATABASE=smart
      - MYSQL_ROOT_PASSWORD=123456
    ports:
      - "3307:3306"
    expose:
      - "3306"
    volumes:
      - mysql-data:/var/lib/mysql  # Thêm dòng này để lưu trữ dữ liệu MySQL
      
  backend-spring:
    build:
      context: ../smart-java-spring-final
      dockerfile: Dockerfile
    ports:
      - "8088:8080"
    environment:
      - SPRING_DATASOURCE_URL=jdbc:mysql://db-mysql:3306/smart
      - SPRING_DATASOURCE_USERNAME=root
      - SPRING_DATASOURCE_PASSWORD=123456
      - SPRING_JPA_HIBERNATE_DDL_AUTO=update
      - PROJECT_UPLOAD_FILE_BASE_URI=file:///uploads/
    depends_on:
      - db-mysql
    volumes:
      - ../upload-docker/:/uploads/

volumes:
  mysql-data:  # Khai báo volume này
